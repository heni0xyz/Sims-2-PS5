cmake_minimum_required(VERSION 3.25)
project(sims2_ps5_game LANGUAGES C CXX)
if(NOT PROSPERO)
    message(FATAL_ERROR "Use the public PS5 SDK toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PS5_FOUNDATION "${ROOT}/../PS5_Sims2Independent" CACHE PATH "Title generated by the pinned PS5_VulkanTemplate")
set(PS5_VULKAN_DIR "${ROOT}/../PS5_Vulkan" CACHE PATH "Pinned PS5 Vulkan driver checkout")
set(GLSLANG_SOURCE "${ROOT}/../glslang" CACHE PATH "glslang 15.1.0 with pinned External sources")
set(GLSLANG_BUILD "${ROOT}/out/build/ps5-glslang" CACHE PATH "PS5 static glslang build")
set(TOML_INCLUDE "${ROOT}/out/build/host-tools/_deps/tomlplusplus-src/include" CACHE PATH "tomlplusplus v3.4.0 headers")
set(SIMS2_BUILD_ID "" CACHE STRING "Optional explicit identifier; normally derived from build inputs")
option(SIMS2_DIAGNOSTIC_LOG "Keep verbose HLE diagnostics" ON)
option(SIMS2_PERFORMANCE_LOG "Emit low-frequency renderer performance reports" OFF)
option(SIMS2_OBJECT_WAIT_CHANNELS "Object-specific dispatcher wake channels" OFF)
option(SIMS2_DEEP_PROFILE "Detailed diagnostic CPU/wait/frame/GPU telemetry" OFF)
option(SIMS2_GUEST_PROFILE "Diagnostic sampled guest/kernel call wall times" OFF)
option(SIMS2_FAST_HANDLES "Cache hot kernel handles with close/publication invalidation" ON)
if(SIMS2_GUEST_PROFILE AND NOT SIMS2_PERFORMANCE_LOG)
    message(FATAL_ERROR "Guest call profiling requires performance reports")
endif()
option(SIMS2_VULKAN_STARTUP_PROBE "Drain early GPU operation groups for a diagnostic experiment" OFF)
if(SIMS2_VULKAN_STARTUP_PROBE AND NOT SIMS2_DIAGNOSTIC_LOG)
    message(FATAL_ERROR "Startup completion probes require diagnostic logging")
endif()
if(NOT SIMS2_BUILD_ID)
    file(GLOB_RECURSE identity_sources CONFIGURE_DEPENDS "${ROOT}/platform/ps5/*" "${ROOT}/runtime-sdk/runtime/*.cpp"
         "${ROOT}/runtime-sdk/runtime/*.h" "${ROOT}/src/*.cpp" "${ROOT}/src/*.h"
         "${ROOT}/generated/independent/*.cpp" "${ROOT}/generated/independent/*.h"
         "${ROOT}/config/*.toml" "${ROOT}/ps5/*.txt" "${ROOT}/repositories.lock.json")
    string(TOUPPER "${CMAKE_BUILD_TYPE}" identity_configuration)
    # Optimization and logging variants must never share a runtime identity.
    set(identity "${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}-${CMAKE_BUILD_TYPE}-${CMAKE_CXX_FLAGS}-${CMAKE_CXX_FLAGS_${identity_configuration}}-${CMAKE_C_FLAGS}-${CMAKE_C_FLAGS_${identity_configuration}}-${SIMS2_DIAGNOSTIC_LOG}-${SIMS2_PERFORMANCE_LOG}-${SIMS2_VULKAN_STARTUP_PROBE}-${SIMS2_GUEST_PROFILE}-${SIMS2_FAST_HANDLES}-${SIMS2_DEEP_PROFILE}-${SIMS2_OBJECT_WAIT_CHANNELS}")
    foreach(source IN LISTS identity_sources)
        file(SHA256 "${source}" sha)
        file(RELATIVE_PATH relative_source "${ROOT}" "${source}")
        string(APPEND identity "${relative_source}:${sha};")
    endforeach()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${identity_sources})
    string(SHA256 SIMS2_BUILD_ID "${identity}")
    string(SUBSTRING "${SIMS2_BUILD_ID}" 0 24 SIMS2_BUILD_ID)
endif()
if(NOT SIMS2_BUILD_ID MATCHES "^[a-zA-Z0-9_.-]+$")
    message(FATAL_ERROR "Build identification must be one diagnostic token")
endif()
foreach(required "${PS5_FOUNDATION}/ps5/tools/link-title.sh" "${TOML_INCLUDE}/toml++/toml.hpp"
                 "${GLSLANG_SOURCE}/glslang/Public/ShaderLang.h")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing prerequisite: ${required}")
    endif()
endforeach()
add_subdirectory(../guest guest)
file(GLOB core CONFIGURE_DEPENDS "${ROOT}/runtime-sdk/runtime/kernel/*.cpp" "${ROOT}/runtime-sdk/runtime/cpu/*.cpp"
     "${ROOT}/runtime-sdk/runtime/xdk/*.cpp" "${ROOT}/runtime-sdk/runtime/gpu/*.cpp" "${ROOT}/src/game/*.cpp")
list(REMOVE_ITEM core "${ROOT}/runtime-sdk/runtime/kernel/memory.cpp" "${ROOT}/runtime-sdk/runtime/gpu/d3d12.cpp")
file(WRITE "${CMAKE_BINARY_DIR}/image_hash.cpp" "extern const char kImageSha1[] = \"16f9e41ce26c315a758f25d43ba1fa9d3809d1ad\";\n")
add_library(sims2-runtime OBJECT ${core}
    "${ROOT}/runtime-sdk/runtime/core/guest_profile.cpp"
    "${ROOT}/runtime-sdk/runtime/core/deep_profile.cpp"
    "${ROOT}/runtime-sdk/runtime/app/game_files.cpp" "${ROOT}/runtime-sdk/runtime/app/settings.cpp"
    "" "${CMAKE_BINARY_DIR}/image_hash.cpp"
    "${ROOT}/platform/ps5/app.cpp" "${ROOT}/platform/ps5/main.cpp"
    "${ROOT}/platform/ps5/log.cpp" "${ROOT}/platform/ps5/video.cpp"
    "${ROOT}/platform/ps5/filesystem.cpp"
    "${ROOT}/platform/ps5/installer.cpp"
    "${PS5_FOUNDATION}/ps5/src/platform.c" "${PS5_FOUNDATION}/ps5/third_party/volk/volk.c")
target_include_directories(sims2-runtime PRIVATE "${ROOT}/runtime-sdk/runtime" "${ROOT}/src"
    "${ROOT}/generated/independent" "${ROOT}/platform/ps5" "${PS5_PLATFORM_HEADERS}"
    "${PS5_FOUNDATION}/ps5/src" "${PS5_FOUNDATION}/ps5/third_party/volk"
    "${PS5_FOUNDATION}/external" "${TOML_INCLUDE}" "${GLSLANG_SOURCE}")
target_compile_definitions(sims2-runtime PRIVATE VK_NO_PROTOTYPES "SIMS2_BUILD_ID=\"${SIMS2_BUILD_ID}\"")
target_compile_options(sims2-runtime PRIVATE -march=x86-64-v2 -fPIC -fno-strict-aliasing
    -fdenormal-fp-math=ieee -fno-omit-frame-pointer -ffunction-sections -fdata-sections
    "SHELL:-include \"${PS5_FOUNDATION}/ps5/third_party/volk/volk.h\"")
if(SIMS2_OBJECT_WAIT_CHANNELS)
    target_compile_definitions(sims2-runtime PRIVATE XBE_OBJECT_WAIT_CHANNELS=1)
endif()
if(SIMS2_DEEP_PROFILE)
    target_compile_definitions(sims2-runtime PRIVATE XBE_DEEP_PROFILE=1)
endif()
if(SIMS2_FAST_HANDLES)
    target_compile_definitions(sims2-runtime PRIVATE XBE_FAST_HANDLES=1)
endif()
if(SIMS2_GUEST_PROFILE)
    target_compile_definitions(sims2-runtime PRIVATE XBE_GUEST_PROFILE=1)
endif()
if(SIMS2_PERFORMANCE_LOG)
    target_compile_definitions(sims2-runtime PRIVATE SIMS2_PERFORMANCE_LOG)
endif()
if(SIMS2_VULKAN_STARTUP_PROBE)
    target_compile_definitions(sims2-runtime PRIVATE SIMS2_VULKAN_STARTUP_PROBE)
endif()
if(NOT SIMS2_DIAGNOSTIC_LOG)
    target_compile_definitions(sims2-runtime PRIVATE NDEBUG)
endif()
set(shader_libraries "${GLSLANG_BUILD}/glslang/libglslang.a"
    "${GLSLANG_BUILD}/SPIRV/libSPIRV.a" "${GLSLANG_BUILD}/glslang/libglslang-default-resource-limits.a"
    "${GLSLANG_BUILD}/External/spirv-tools/source/opt/libSPIRV-Tools-opt.a"
    "${GLSLANG_BUILD}/External/spirv-tools/source/libSPIRV-Tools.a")
foreach(library IN LISTS shader_libraries)
    if(NOT EXISTS "${library}")
        message(FATAL_ERROR "Missing PS5 shader compiler archive: ${library}")
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/runtime-objects.txt" CONTENT "$<JOIN:$<TARGET_OBJECTS:sims2-runtime>,\n>\n")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/link-archives.txt" CONTENT "$<TARGET_FILE:sims2-guest>\n$<TARGET_FILE:sims2-ps5-memory>\n$<TARGET_FILE:sims2-ps5-services>\n$<JOIN:${shader_libraries},\n>\n")
file(WRITE "${CMAKE_BINARY_DIR}/build-id.txt" "${SIMS2_BUILD_ID}\n")
file(WRITE "${CMAKE_BINARY_DIR}/samples.txt" "sims2independent\n")
# The established title helper owns CRT, RADV, platform wraps, ELF conversion,
# signing, import validation, libc.prx, notices and package assembly.
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/eboot.elf"
    COMMAND bash "${PS5_FOUNDATION}/ps5/tools/link-title.sh" "${CMAKE_BINARY_DIR}"
        "${PS5_VULKAN_DIR}" "${PS5_PAYLOAD_SDK}" "$<TARGET_OBJECTS:sims2-runtime>"
        --start-group "$<TARGET_FILE:sims2-guest>" "$<TARGET_FILE:sims2-ps5-memory>"
        "$<TARGET_FILE:sims2-ps5-services>" ${shader_libraries} --end-group
    DEPENDS sims2-runtime sims2-guest sims2-ps5-memory sims2-ps5-services ${shader_libraries}
        "${PS5_FOUNDATION}/ps5/tools/link-title.sh"
    COMMAND_EXPAND_LISTS VERBATIM)
add_custom_target(sims2-title ALL DEPENDS "${CMAKE_BINARY_DIR}/eboot.elf")
