cmake_minimum_required(VERSION 3.25)
project(sims2_ps5_guest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT PROSPERO)
    message(FATAL_ERROR "Configure with the public SDK's toolchain/prospero.cmake")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PS5_PLATFORM_HEADERS "" CACHE PATH "platform/include exported at the template SDK pin")
if(NOT EXISTS "${PS5_PLATFORM_HEADERS}/ps5platform/shm.h")
    message(FATAL_ERROR "PS5_PLATFORM_HEADERS must contain the pinned platform headers")
endif()
file(STRINGS "${ROOT}/config/game.toml" partitions REGEX "^partitions *=" )
string(REGEX MATCH "[0-9]+" partitions "${partitions}")
math(EXPR last "${partitions}-1")
set(guest "${ROOT}/generated/independent/x86_func_mapping.cpp")
foreach(i RANGE ${last})
    list(APPEND guest "${ROOT}/generated/independent/x86_recomp.${i}.cpp")
endforeach()
foreach(source IN LISTS guest)
    if(NOT EXISTS "${source}")
        message(FATAL_ERROR "Run the pinned host xbe_recomp on config/game.toml first")
    endif()
endforeach()
add_library(sims2-guest STATIC ${guest})
target_include_directories(sims2-guest PRIVATE "${ROOT}/runtime-sdk/runtime" "${ROOT}/generated/independent")
target_compile_options(sims2-guest PRIVATE -fno-strict-aliasing -Wno-unused-label
    -Wno-parentheses-equality -Wno-unused-variable)
add_library(sims2-ps5-memory STATIC "${ROOT}/runtime-sdk/runtime/platform/ps5_memory.cpp"
    "${ROOT}/runtime-sdk/runtime/kernel/memory.cpp" "${ROOT}/platform/ps5/contracts.cpp")
target_include_directories(sims2-ps5-memory PRIVATE "${ROOT}/runtime-sdk/runtime"
    "${ROOT}/platform/ps5" "${PS5_PLATFORM_HEADERS}")
add_library(sims2-ps5-services STATIC "${ROOT}/platform/ps5/services.cpp" "${ROOT}/platform/ps5/title.cpp")
target_include_directories(sims2-ps5-services PRIVATE "${ROOT}/runtime-sdk/runtime"
    "${ROOT}/platform/ps5" "${PS5_FOUNDATION}/ps5/src" "${PS5_PLATFORM_HEADERS}")
foreach(target sims2-guest sims2-ps5-memory sims2-ps5-services)
    target_compile_options(${target} PRIVATE -march=x86-64-v2 -fno-omit-frame-pointer -fdenormal-fp-math=ieee
        -ffunction-sections -fdata-sections)
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
# Final title linkage is owned by the template/RADV recipe in the parent target.
