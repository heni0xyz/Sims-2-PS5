# Native Windows form of the MIT SDK's Linux null-backend harness.
# Original source selection: sdk/harness/build.sh, Copyright 2026 TGP482.
cmake_minimum_required(VERSION 3.25)
project(sims2_independent_windows_headless LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Use tools/build_runtime_headless.sh on Linux")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(RT "${ROOT}/runtime-sdk/runtime")
set(HARNESS "${ROOT}/runtime-sdk/harness")
set(GEN "${ROOT}/generated/independent")
set(SIMS2_WINDOWS_BACKEND "NULL" CACHE STRING "NULL logic probe or VULKAN offscreen renderer")
set_property(CACHE SIMS2_WINDOWS_BACKEND PROPERTY STRINGS NULL VULKAN)
option(SIMS2_WINDOWS_RENDER_THREAD "Exercise the same recording backend as PS5" OFF)
if(SIMS2_WINDOWS_RENDER_THREAD)
    # Override only the HLE source that selects the backend. Guest compilation
    # and the existing immutable generated-object cache are unaffected.
    set_property(SOURCE "${RT}/xdk/d3d8.cpp" APPEND PROPERTY COMPILE_OPTIONS -UXBE_RENDER_THREAD -DXBE_RENDER_THREAD=1)
endif()
if(NOT SIMS2_WINDOWS_BACKEND MATCHES "^(NULL|VULKAN)$")
    message(FATAL_ERROR "Select NULL or VULKAN")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB runtime CONFIGURE_DEPENDS "${RT}/kernel/*.cpp" "${RT}/cpu/*.cpp"
     "${RT}/xdk/*.cpp" "${RT}/gpu/*.cpp" "${RT}/core/*.cpp" "${RT}/audio/*.cpp")
foreach(source core/input.cpp gpu/d3d12.cpp gpu/vulkan.cpp gpu/spirv.cpp
               gpu/texturedumper.cpp audio/sdl_audio.cpp audio/xaudio2.cpp)
    list(REMOVE_ITEM runtime "${RT}/${source}")
endforeach()
file(GLOB support CONFIGURE_DEPENDS "${ROOT}/src/*/*.cpp" "${HARNESS}/*.cpp")
if(SIMS2_WINDOWS_BACKEND STREQUAL "NULL")
    list(REMOVE_ITEM support "${ROOT}/src/game/install_display.cpp")
endif()
if(SIMS2_WINDOWS_BACKEND STREQUAL "VULKAN")
    enable_language(C)
    set(PS5_FOUNDATION "${ROOT}/../PS5_Sims2Independent" CACHE PATH "Template containing public volk/Vulkan headers")
    set(GLSLANG_SOURCE "" CACHE PATH "Pinned public glslang source")
    if(NOT EXISTS "${GLSLANG_SOURCE}/glslang/Public/ShaderLang.h")
        message(FATAL_ERROR "Supply the recorded glslang source checkout")
    endif()
    foreach(flag BUILD_SHARED_LIBS GLSLANG_TESTS ENABLE_GLSLANG_BINARIES ENABLE_SPVREMAPPER ENABLE_PCH)
        set(${flag} OFF CACHE BOOL "Native diagnostic compiler selection" FORCE)
    endforeach()
    # HLSL requires SPIR-V legalization. Match the PS5 compiler's complete path.
    set(BUILD_EXTERNAL ON CACHE BOOL "Pinned SPIR-V tools source" FORCE)
    set(ENABLE_OPT ON CACHE BOOL "Required HLSL SPIR-V legalization" FORCE)
    set(SPIRV_SKIP_TESTS ON CACHE BOOL "Library build; no dependency test executables" FORCE)
    set(SPIRV_SKIP_EXECUTABLES ON CACHE BOOL "Static shader compiler libraries" FORCE)
    set(ENABLE_HLSL ON CACHE BOOL "HLSL input required by the Xbox renderer" FORCE)
    set(ENABLE_SPIRV ON CACHE BOOL "Vulkan shader output" FORCE)
    add_subdirectory("${GLSLANG_SOURCE}" glslang)
    # The pinned validator relied on a transitive STL include removed by
    # libc++ 23. Supply its actual required header for this source only,
    # without modifying the shared dependency checkout.
    set_property(SOURCE "${GLSLANG_SOURCE}/External/spirv-tools/source/val/validate_cfg.cpp"
        TARGET_DIRECTORY SPIRV-Tools-static PROPERTY COMPILE_OPTIONS "-include;iterator")
    list(APPEND runtime "${RT}/gpu/vulkan.cpp" "${RT}/gpu/spirv.cpp")
    list(REMOVE_ITEM support "${HARNESS}/null_backend.cpp")
    add_library(independent-volk OBJECT "${PS5_FOUNDATION}/ps5/third_party/volk/volk.c")
    target_include_directories(independent-volk PRIVATE "${PS5_FOUNDATION}/external")
endif()
file(GLOB guest CONFIGURE_DEPENDS "${GEN}/x86_recomp.*.cpp" "${GEN}/x86_func_mapping.cpp")
list(LENGTH guest guest_count)
if(NOT guest_count EQUAL 65)
    message(FATAL_ERROR "Regenerate the reviewed independent configuration first")
endif()
list(REMOVE_ITEM support "${HARNESS}/main_test.cpp")
add_library(independent-host OBJECT ${runtime} ${support})
add_library(independent-entry OBJECT "${HARNESS}/main_test.cpp")
add_library(independent-guest OBJECT ${guest})
foreach(target independent-host independent-guest independent-entry)
    target_include_directories(${target} PRIVATE "${ROOT}/src" "${RT}" "${GEN}" "${HARNESS}")
    target_compile_options(${target} PRIVATE -O1 -g1 -fno-strict-aliasing -fms-extensions -march=x86-64-v2)
    target_compile_definitions(${target} PRIVATE SIMS2_HEADLESS_RUNTIME=1 XBE_RAW_VERTICES=0
        XBE_RENDER_THREAD=0 WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
endforeach()
target_compile_options(independent-host PRIVATE "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/compat.h\"")
target_compile_options(independent-entry PRIVATE "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/compat.h\"")
add_executable(Sims2Independent $<TARGET_OBJECTS:independent-host> $<TARGET_OBJECTS:independent-guest> $<TARGET_OBJECTS:independent-entry>)
target_link_libraries(Sims2Independent PRIVATE ws2_32 user32)
if(SIMS2_WINDOWS_BACKEND STREQUAL "VULKAN")
    target_compile_definitions(independent-host PRIVATE XBE_VULKAN_HOST=1)
    target_include_directories(independent-host PRIVATE "${PS5_FOUNDATION}/external"
        "${PS5_FOUNDATION}/ps5/third_party/volk" "${GLSLANG_SOURCE}")
    target_compile_options(independent-host PRIVATE "SHELL:-include \"${PS5_FOUNDATION}/ps5/third_party/volk/volk.h\"")
    target_sources(Sims2Independent PRIVATE $<TARGET_OBJECTS:independent-volk>)
    target_link_libraries(Sims2Independent PRIVATE glslang SPIRV glslang-default-resource-limits)
    set_target_properties(Sims2Independent PROPERTIES OUTPUT_NAME Sims2IndependentVulkan)
    add_executable(Sims2InstallerProbe "${ROOT}/tests/installer_vulkan_probe.cpp"
        $<TARGET_OBJECTS:independent-host> $<TARGET_OBJECTS:independent-guest> $<TARGET_OBJECTS:independent-volk>)
    target_include_directories(Sims2InstallerProbe PRIVATE "${ROOT}/src" "${RT}" "${GEN}")
    target_compile_features(Sims2InstallerProbe PRIVATE cxx_std_20)
    target_compile_options(Sims2InstallerProbe PRIVATE -fms-extensions "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/compat.h\"")
    target_link_libraries(Sims2InstallerProbe PRIVATE ws2_32 user32 glslang SPIRV glslang-default-resource-limits)
endif()
