cmake_minimum_required(VERSION 3.20)
project(sims2_console_generator LANGUAGES C CXX)
get_filename_component(SIMS2_PROJECT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SIMS2_ZYDIS_SOURCE "${SIMS2_PROJECT}/out/host/_deps/zydis-src" CACHE PATH "Pinned Zydis v4.1.1 source")
set(SIMS2_TOMLPLUSPLUS_SOURCE "${SIMS2_PROJECT}/out/host/_deps/tomlplusplus-src" CACHE PATH "Pinned tomlplusplus v3.4.0 source")
if(NOT EXISTS "${SIMS2_ZYDIS_SOURCE}/CMakeLists.txt" OR NOT EXISTS "${SIMS2_TOMLPLUSPLUS_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Provide the pinned public dependency source directories; no network fetch in this build")
endif()
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
add_subdirectory("${SIMS2_ZYDIS_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/zydis")
add_subdirectory("${SIMS2_TOMLPLUSPLUS_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/tomlplusplus")
set(sdk "${SIMS2_PROJECT}/sdk")
file(GLOB generator_sources CONFIGURE_DEPENDS "${sdk}/tools/recomp/*.cpp")
set_source_files_properties("${sdk}/tools/recomp/main.cpp" PROPERTIES
  COMPILE_DEFINITIONS "main=sims2_xbe_recomp_main")
add_library(sims2-console-generator STATIC ${generator_sources} ../generator.cpp)
target_compile_features(sims2-console-generator PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  # PS5's compiler defaults to exceptions off. The unchanged SDK parser and
  # tomlplusplus use exceptions, as in the working owner's builder recipe.
  target_compile_options(sims2-console-generator PRIVATE -fexceptions -frtti)
endif()
target_include_directories(sims2-console-generator PRIVATE "${sdk}/runtime" "${sdk}/tools/recomp")
target_link_libraries(sims2-console-generator PRIVATE Zydis tomlplusplus::tomlplusplus)
option(SIMS2_GENERATOR_HOST_TEST "Build host CLI for in-process generation parity" OFF)
if(SIMS2_GENERATOR_HOST_TEST)
  add_executable(sims2-generator-host-test ../generator_probe.cpp)
  target_link_libraries(sims2-generator-host-test PRIVATE sims2-console-generator)
endif()
